<script>
  (function () {
    const LINKEDIN_URL = "https://www.linkedin.com/in/kelley-bennett-742755b/details/recommendations/";

    document.querySelectorAll("[data-rec-linkedin]").forEach((link) => {
      link.setAttribute("href", LINKEDIN_URL);
      link.setAttribute("target", "_blank");
      link.setAttribute("rel", "noopener noreferrer");
    });

    document.querySelectorAll(".rec-toggle").forEach((btn) => {
      btn.addEventListener("click", () => {
        const card = btn.closest(".rec-card");
        const full = card.querySelector(".rec-full");
        const expanded = btn.getAttribute("aria-expanded") === "true";
        btn.setAttribute("aria-expanded", String(!expanded));
        full.hidden = expanded;
        btn.querySelector(".rec-toggle-label").textContent = expanded
          ? "Read the full recommendation"
          : "Show less";
      });
    });
  })();
</script>
